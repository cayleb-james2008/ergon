   * @param {{task: string, checker: string, round: number, result: string, exit_code?: number, artefact_hashes?: string[], doneWhenMode: string, briefHashes: string[]}} v
   */
  function verdictGate(v) {
    if (v.doneWhenMode !== 'command' || v.result === 'invalid-check') return null
    const last = store.runsOf(v.task, v.checker, v.round).at(-1)
    if (last === undefined) return 'no-check-run'
    if (v.exit_code !== last.exit_code) return 'exit-mismatch'
    const allowed = new Set(v.briefHashes.map(hexOf))
    if ((v.artefact_hashes ?? []).some(h => !allowed.has(hexOf(h)))) return 'hash-mismatch'
    if (v.result === 'pass' && last.exit_code !== 0) return 'pass-with-nonzero'
    return null
  }
