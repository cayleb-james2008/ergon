# plugin-add / plugin-remove / restore-* take arguments; they are listed with wildcards because the
# wrapper validates every argument itself (sha256 of the copied tarball, name regexes, no passthrough).
# To make installs require Cayleb's password instead, delete the PLUGIN line and use:
#   pkexec --user fleet /usr/local/bin/dsh-fleet-admin plugin-add <sha256> <file.tgz>

                        /usr/local/bin/dsh-fleet-admin probe-access, \
                        /usr/local/bin/dsh-fleet-admin init-profile, \
                        /usr/local/bin/dsh-fleet-admin install-dsh 0.2.2-alpha.2, \

                        /usr/local/bin/dsh-fleet-admin integrity-drill restore

                         /usr/local/bin/dsh-fleet-admin plugin-remove *
